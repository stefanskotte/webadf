## Rulings (where the spec is silent or the code decides)

- **R1 Drawn order in LIBRARY:.** The server returns the spec's order (collections by `sortKey, id`; then uncategorized titles by `sortTitle, id`; a collection's titles by its own `sortKey`). The pane draws through `sortFsEntries` like every Files listing: collections always first, then the other folders, then files; within each group the pane's column sort (Name ▲ by default). The spec's "in the user's Library order" therefore shows only as the tie order. Cost if wrong: one comparator.
- **R2 LIBRARY: segments are the server's business.** Collection ids and game ids are both UUID-shaped, so only the database can tell `LIBRARY:/<x>/…` apart. The client stops parsing LIBRARY: segments: it uses `Listing.disk` (report targets), `Listing.diskBase` ("new in vN" marks, the drop folder), `Listing.library` (shelf rules), and an unreadable disk's id comes back as the `blob_unavailable` refusal's `detail` (for its strip).
- **R3 Collection names.** The Library's rules (trimmed, 1–80 characters). The Library page has **no** clash rule today, so the spec's "the Library's words" do not exist: Files refuses a case-insensitive clash with another collection as `name_exists` "A collection with that name already exists." /** "3 added · 1 refused", and "· 2 working" while any row is still hashing or uploading. A disk the library already
```ts
 *  had is added too (Files 2 §5: one "Added"; the server only skipped the upload). A count of nothing is left out. */
export function traySummary(rows: LibRow[]): string {
  let added = 0, refused = 0, working = 0;
  for (const r of rows) {
    if (r.state === 'done' || r.state === 'deduped') added++;
    else if (r.state === 'failed') refused++;
    else working++;
  }
  return [added && `${added} added`, refused && `${refused} refused`, working && `${working} working`].filter(Boolean).join(' · ');
}

/** One row's state word and tone: Hashing… / Uploading… / Added / Refused (— why). */
export function trayState(r: LibRow): { word: string; tone: 'busy' | 'done' | 'wait' | 'fail' } {
  switch (r.state) {
    case 'hashing': return { word: 'Hashing…', tone: 'busy' };
    case 'uploading': return { word: 'Uploading…', tone: 'busy' };
    case 'done': case 'deduped': return { word: 'Added', tone: 'done' };
    case 'failed': return { word: r.note ? `Refused — ${r.note}` : 'Refused', tone: 'fail' };
  }
}
```

```ts
    .innerJoin(games, and(eq(games.id, collectionGames.gameId), eq(games.orgId, ctx.orgId), isLocal(games)))

export async function placeInLibrary(ctx: FsCtx, segments: readonly string[]): Promise<LibraryPlace | null> {
  if (!segments.length) return { level: 'root', collection: null, prefix: [] };
  const collection = await collectionOf(ctx, segments[0]);
  if (collection && segments.length === 1) return { level: 'collection', collection, prefix: [collection.id] };
  const prefix = collection ? [collection.id] : [];
  const [gameId, diskId, ...names] = segments.slice(prefix.length);
  if (collection && !(await isFiledIn(ctx, collection.id, gameId))) return null;
  if (!diskId) return { level: 'title', collection, prefix, gameId };
  return { level: 'disk', collection, prefix, gameId, diskId, names };
}
```

```ts
/** A LIBRARY: path inside a disk. `canon` is the same place without its collection prefix: what two paths of one disk
 *  are compared by (a cycle, an entry inside a folder), whichever form each came in (R2). */
type Loc = { path: FsPath; canon: FsPath; gameId: string; diskId: string; names: string[] };

async function locate(ctx: FsCtx, raw: string): Promise<Loc> {
  const path = parsePath(raw);
  if (!path) refuse('not_found', 404);
  if (path.device !== 'LIBRARY') refuse('unsupported', 400, `${path.device}: is not available yet`);
```

```ts
/** A path's place on its disk ('' for the disk itself, '/'-joined names below it); null outside `base` or without one.
 *  `base` is the listing's diskBase (Files 2 R2): a disk reached through a collection has one more segment. */
export function diskRelative(path: string, base: string | null | undefined): string | null {
  if (!base) return null;
  if (path === base) return '';
  return path.startsWith(`${base}/`) ? path.slice(base.length + 1) : null;
}

export function withAddedIn(entries: FsEntry[], marks: ReadonlyMap<string, number>, base: string | null | undefined): FsEntry[] {
  if (marks.size === 0) return entries;
  return entries.map((e) => {
    const rel = diskRelative(e.path, base);
    const seq = rel === null ? undefined : marks.get(rel);
    return seq === undefined ? e : { ...e, addedIn: seq };
  });
}
```

```ts
// Add to library (increment 2, R8), shared by the op and by filing onto LIBRARY: (Files 2 R4). SERVER ONLY.
import { parsePath } from './path';
import { refuse, Refused, type Added, type OpCtx, type OpItem } from './driver';
import { stashEntryOf, STILL_ARRIVING } from './stash-driver';
import { addUploadToLibrary } from '@/lib/stash-library';
import { libraryEligible } from '@/lib/from-amiga-ui';

/** Per item: a stored disk image in STASH: itself goes through addUploadToLibrary; nothing else does. */
export async function addDiskImages(ctx: OpCtx, sources: string[]): Promise<{ items: OpItem[]; added: Added[] }> {
  const items: OpItem[] = [];
  const added: Added[] = [];
  for (const source of sources) {
    try {
      const p = parsePath(source);
      if (!p) refuse('not_found', 404);
      if (p.device !== 'STASH') refuse('unsupported', 400, 'only files in STASH: are added to the library');
      const at = await stashEntryOf(ctx, p);
      if (!at) refuse('not_found', 404);
      if (at.in === 'archive') refuse('unsupported', 400, 'copy it out of the archive first');
      const row = at.row;
      if (row?.status === 'receiving') refuse('unsupported', 409, STILL_ARRIVING);
      if (!row || row.kind !== 'file') refuse('not_a_disk_image', 422, 'a folder');
      if (row.status !== 'stored') refuse('not_found', 404);
      if (!libraryEligible(row.path, row.sizeBytes)) refuse('not_a_disk_image', 422);
      const r = await addUploadToLibrary(ctx.orgId, row.id);
      if (!r.ok) {
        if (r.error === 'not_found') refuse('not_found', 404);
        if (r.error === 'ingest_failed') refuse('failed', 502, 'the library could not take it');
        refuse('not_a_disk_image', 422, r.reason);
      }
      added.push({ source, gameId: r.gameId, diskId: r.diskId });
      items.push({ source, ok: true });
    } catch (e) {
      if (!(e instanceof Refused)) throw e;
      items.push({ source, ok: false, refusal: e.refusal });
    }
  }
  return { items, added };
}
```

```ts
// Delete of a collection; Copy and Move of titles between the collections and the root; disk images from STASH: into

/** A LIBRARY: destination above the disks (the root, a collection, a title's disk list); null for anything else. */
export async function shelfDest(ctx: FsCtx, p: FsPath): Promise<ShelfDest | null> {
  if (p.device !== 'LIBRARY') return null;
  const at = await placeInLibrary(ctx, p.segments);
  return at && at.level !== 'disk' ? at : null;
}

/** A LIBRARY: row above a disk's folders (a collection, a title, a disk itself): what Rename and Delete answer here. */
export async function shelfItem(ctx: FsCtx, p: FsPath): Promise<LibraryPlace | null> {
  if (p.device !== 'LIBRARY') return null;
  const at = await placeInLibrary(ctx, p.segments);
  return at && at.level !== 'root' && !(at.level === 'disk' && at.names.length > 0) ? at : null;
}

/** The collection rules (R3): trimmed, 1-80 characters, and no other collection of this org by that name (any case). */
async function cleanName(ctx: FsCtx, name: string, except: string | null): Promise<string> {
  const problem = collectionNameProblem(name);
  if (problem) refuse('bad_name', 400, problem);
  const clean = name.trim();
  const taken = (await listCollections(ctx.orgId)).some((c) => c.id !== except && c.name.trim().toLowerCase() === clean.toLowerCase());
  if (taken) refuse('name_exists', 409, COLLECTION_EXISTS);
  return clean;
}

export async function makeCollection(ctx: OpCtx, dest: ShelfDest, name: string): Promise<Outcome> {
  if (dest.level === 'collection') refuse('unsupported', 400, NO_NESTED);
  if (dest.level === 'title') refuse('unsupported', 400, 'Open a disk or STASH: to make a folder in it');
  const made = await createCollection(ctx.orgId, await cleanName(ctx, name, null));
  return { kind: 'done', items: [{ source: libPath(made.id), ok: true }] };
}

export async function renameOnShelf(ctx: OpCtx, source: string, at: LibraryPlace, name: string): Promise<Outcome> {
  if (at.level !== 'collection') refuse('unsupported', 400, TITLES_ON_LIBRARY_PAGE);
  const clean = await cleanName(ctx, name, at.collection.id);
  if (!(await renameCollection(ctx.orgId, at.collection.id, clean))) refuse('not_found', 404);
  return { kind: 'done', items: [{ source, ok: true }] };
}

/** Each collection deleted (its titles stay in the library); a title or a disk row is refused, per item. */
export async function deleteOnShelf(ctx: OpCtx, sources: string[]): Promise<Outcome> {
  const items: OpItem[] = [];
  for (const source of sources) {
    const p = parsePath(source);
    const at = p ? await shelfItem(ctx, p) : null;
    if (!at) { items.push(gone(source)); continue; }
    if (at.level !== 'collection') { items.push(fail(source, TITLES_ON_LIBRARY_PAGE)); continue; }
    items.push((await deleteCollection(ctx.orgId, at.collection.id)) ? { source, ok: true } : gone(source));
  }
  return { kind: 'done', items };

type Title = { source: string; gameId: string; from: string | null };

/**
 * Copy or Move onto the root or a collection (D4, R4, R5). Titles change membership; disk images from STASH: are added
 * to the library and filed. A title is never duplicated: one already in the target is skipped and left where it was.
 */
export async function fileOnShelf(ctx: OpCtx, req: CopyMove, dest: ShelfDest): Promise<Outcome> {
  if (dest.level === 'title') {
    if (!(await titleExists(ctx, dest.gameId))) refuse('not_found', 404);   // a deleted collection reads as a title id
    refuse('unsupported', 400, FILE_ON_SHELF);
  }
  const to = dest.level === 'collection' ? dest.collection.id : null;
  const results = new Map<string, OpItem>();
  const titles: Title[] = [];
  const images: string[] = [];
  for (const source of req.sources) {
    const p = parsePath(source);
    if (p?.device === 'STASH') { images.push(source); continue; }
    const at = p?.device === 'LIBRARY' ? await placeInLibrary(ctx, p.segments) : null;
    if (p?.device === 'LIBRARY' && !at) { results.set(source, gone(source)); continue; }
    if (at?.level === 'title') { titles.push({ source, gameId: at.gameId, from: at.collection?.id ?? null }); continue; }
    results.set(source, fail(source, at?.level === 'collection' ? NO_COLLECTION_IN_COLLECTION : ONLY_TITLES_FILED));
  }
  await fileTitles(ctx, req.op, titles, to, results);
  const added = images.length ? await fileImages(ctx, req.op, images, to, results) : [];
  return { kind: 'done', items: req.sources.map((s) => results.get(s)!), ...(added.length ? { added } : {}) };
}

async function fileTitles(ctx: OpCtx, op: 'copy' | 'move', titles: Title[], to: string | null, results: Map<string, OpItem>) {
  if (!titles.length) return;
  if (to === null) {
    // To the root: a copy has nothing to do; a move takes each out of the collection it was listed under.
    for (const t of titles) {
      if (op === 'copy' || t.from === null) { results.set(t.source, fail(t.source, IN_LIBRARY_ALREADY)); continue; }
      results.set(t.source, (await removeGameFromCollection(ctx.orgId, t.from, t.gameId)) ? { source: t.source, ok: true } : gone(t.source));
    }
    return;
  }
  // Already there (R5; Review Focus 1: also every title listed in `to` itself): skipped, and left in its own collection.
  const already = new Set((await getDb().select({ g: collectionGames.gameId }).from(collectionGames)
    .where(and(eq(collectionGames.collectionId, to), inArray(collectionGames.gameId, titles.map((t) => t.gameId))))).map((r) => r.g));
  const todo: Title[] = [];
  for (const t of titles) {
    if (already.has(t.gameId)) results.set(t.source, skippedItem(t.source));
    else todo.push(t);
  }
  for (const [from, group] of groupBy(todo, (t) => (op === 'move' ? t.from : null))) {
    const r = await moveGamesToCollection(ctx.orgId, group.map((t) => t.gameId), to, from ?? undefined);
    if (!r) { for (const t of group) results.set(t.source, gone(t.source)); continue; }   // the collection went meanwhile
    const missing = new Set(r.missing);
    for (const t of group) results.set(t.source, missing.has(t.gameId) ? gone(t.source) : { source: t.source, ok: true });
  }
}

async function fileImages(ctx: OpCtx, op: 'copy' | 'move', sources: string[], to: string | null, results: Map<string, OpItem>): Promise<Added[]> {
  const out = await addDiskImages(ctx, sources);
  for (const it of out.items) results.set(it.source, it);
  for (const a of out.added) {
    if (to && !(await addGameToCollection(ctx.orgId, to, a.gameId))) results.set(a.source, gone(a.source));
  }
  if (op === 'move') {
    const done = out.added.map((a) => a.source).filter((s) => results.get(s)?.ok);
    if (done.length) {
      try {
        const r = await driverFor('STASH')!.remove!(ctx, done);
        if (r.kind === 'done') {
          for (const x of r.items) if (!x.ok) results.set(x.source, { source: x.source, ok: false, refusal: { code: x.refusal.code, detail: REMOVE_FAILED } });
        }
      } catch (e) {
        if (!(e instanceof Refused)) throw e;
        for (const s of done) results.set(s, { source: s, ok: false, refusal: { code: e.refusal.code, detail: REMOVE_FAILED } });
      }
    }
  }
  return out.added;
}
```

```ts
export async function createBlankDisk(orgId: string, o: {

    if (owned) collectionId = owned.id;
    else if (o.mustFile) return null;
  }
  const bytes = formatVolume({ filesystem: o.filesystem, volumeName: o.volumeName, density: o.density });
  const sha256 = createHash('sha256').update(bytes).digest('hex');
  await diskStore.put(sha256, bytes);
  // (the blobs, entitlements, games and disks inserts, unchanged from the route, with volumeName = o.volumeName)

    await db.insert(collectionGames).values({ collectionId, gameId, sortKey: minSort - 1 }).onConflictDoNothing();
```

```ts
// src/lib/files/create-adf.ts
// Create ADF… in Files (Files 2 §4): a blank image `<Name>.adf` into a STASH: folder (through the STASH: store, so it
// counts toward the org's 500 files / 256 MB and follows the clash setting), or a new title with one blank disk into
// LIBRARY: or a collection. Never in NET:, a title's disk list, a disk or an archive. SERVER ONLY.
import { formatVolume } from '@/lib/adffs/format';
import { createBlankDisk } from '@/lib/blank-disk';
import { parsePath } from './path';
import { libPath, placeInLibrary, titleExists } from './library-paths';
import { writeStash } from './stash-driver';
import { checkName, refuse, type OnClash, type OpCtx, type Outcome } from './driver';
import { GO_UP, NOT_ON_NET } from './shelf-rules';

export type CreateAdfRequest = { dest: string; name: string; filesystem: 'OFS' | 'FFS'; density: 'dd' | 'hd'; onClash: OnClash };
/** The source a STASH: write reports a made image under, before it has a path. */
export const CREATED = 'create-adf';

export async function createAdf(ctx: OpCtx, req: CreateAdfRequest): Promise<Outcome> {
  checkName(req.name);   // R7: an AmigaDOS volume name, in STASH: and LIBRARY: alike
  const p = parsePath(req.dest);
  if (!p) refuse('not_found', 404);
  if (p.device === 'NET') refuse('unsupported', 400, NOT_ON_NET);
  if (p.device === 'STASH') {
    const bytes = formatVolume({ filesystem: req.filesystem, volumeName: req.name, density: req.density });
    const out = await writeStash(ctx, p, [{ source: CREATED, items: [{ rel: [`${req.name}.adf`], kind: 'file', bytes, date: new Date(), protection: 0, comment: null }] }],
      req.onClash, 'copy');
    if (out.kind !== 'done') return out;
    const at = out.placed?.[0]?.path;
    return { kind: 'done', items: out.items.map((i) => (at && i.ok && !i.skipped ? { ...i, source: at } : i)) };
  }
  const at = await placeInLibrary(ctx, p.segments);
  if (!at) refuse('not_found', 404);
  // A foreign or deleted collection id reads as a title id: no such title is not found, not "go up" (§7).
  if (at.level === 'title' && !(await titleExists(ctx, at.gameId))) refuse('not_found', 404);
```

```ts
/** Why Create ADF… cannot make a disk where `l` is (§4), or null. The archive words are refusalWords('read_only',
 *  'archive')'s, written out: words.ts imports this module, so this module never imports words.ts. */
export function createAdfReason(l: Listing | null): string | null {
  if (!l) return 'Open STASH: or LIBRARY: first';
  const device = parsePath(l.path)?.device;
  if (device === 'NET') return NOT_ON_NET;
  if (device === 'STASH') return l.readOnlyReason === 'archive' ? 'Archives are read-only — copy files out of them.' : null;
  return l.library?.level === 'root' || l.library?.level === 'collection' ? null : GO_UP;
}
```
